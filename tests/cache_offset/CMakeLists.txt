cmake_minimum_required(VERSION 3.5)
project(HailoCacheOffsetRegression LANGUAGES CXX)
enable_testing()
add_executable(cache_offset_test cache_offset_test.cpp)
set_property(TARGET cache_offset_test PROPERTY CXX_STANDARD 11)
target_include_directories(cache_offset_test PRIVATE ../../hailort/libhailort/src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cache_offset_test PRIVATE -Wall -Wextra -Wconversion -Werror)
endif()
add_test(NAME cache_offset_regression COMMAND cache_offset_test)
